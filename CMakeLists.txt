cmake_minimum_required(VERSION 3.16)
project(LoadFlowAnalyzer VERSION 3.0 LANGUAGES CXX)
add_executable(load_flow Gauss_Siedel_Load_Flow_Analyser/Gauss_Siedel_Load_Flow_Analyser.cpp)
target_compile_features(load_flow PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(load_flow PRIVATE /W4 /permissive-)
else()
  target_compile_options(load_flow PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_solver tests/test_solver.cpp)
  target_compile_features(test_solver PRIVATE cxx_std_17)
  add_test(NAME numerical_solver COMMAND test_solver)
  add_executable(test_io tests/test_io.cpp)
  target_compile_features(test_io PRIVATE cxx_std_17)
  add_test(NAME csv_and_reports COMMAND test_io)
  if(MINGW AND CMAKE_VERSION VERSION_GREATER_EQUAL 3.22)
    get_filename_component(LOAD_FLOW_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set_tests_properties(numerical_solver csv_and_reports PROPERTIES
      ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${LOAD_FLOW_COMPILER_DIR}")
  endif()
  option(NRLARGE_REQUIRE_PYTHON_TESTS "Require installed NumPy/SciPy for NRlarge tests" OFF)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  set(NRLARGE_PYTHON_READY 1)
  if(Python3_Interpreter_FOUND)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy, scipy"
                    RESULT_VARIABLE NRLARGE_PYTHON_READY OUTPUT_QUIET ERROR_QUIET)
  endif()
  if(Python3_Interpreter_FOUND AND NRLARGE_PYTHON_READY EQUAL 0)
    add_test(NAME nrlarge_economics COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover
             -s tests/economics -p test_*.py -v)
    set_tests_properties(nrlarge_economics PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
    foreach(component model runtime cluster benchmark_cases)
      add_test(NAME nrlarge_${component} COMMAND "${Python3_EXECUTABLE}" -m unittest discover
               -s tests/nrlarge -p test_${component}.py -v)
      set_tests_properties(nrlarge_${component} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
    endforeach()
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import matplotlib, networkx, PIL"
                    RESULT_VARIABLE NRLARGE_GALLERY_READY OUTPUT_QUIET ERROR_QUIET)
    if(NRLARGE_GALLERY_READY EQUAL 0)
      add_test(NAME nrlarge_benchmark_gallery COMMAND "${Python3_EXECUTABLE}" -m unittest discover
               -s tests/nrlarge -p test_benchmark_gallery.py -v)
      set_tests_properties(nrlarge_benchmark_gallery PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
    else()
      message(STATUS "Gallery regression suite NOT enabled: optional benchmark plotting dependencies unavailable")
    endif()
    add_test(NAME nrlarge_gpu COMMAND "${Python3_EXECUTABLE}" -m unittest discover
             -s tests -p test_nrlarge_gpu.py -v)
    set_tests_properties(nrlarge_gpu PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
    set_tests_properties(nrlarge_runtime nrlarge_gpu PROPERTIES RESOURCE_LOCK nrlarge_gpu_device)
    message(STATUS "NRlarge Python suites enabled; optional device/TLS skips are reported by unittest")
  elseif(NRLARGE_REQUIRE_PYTHON_TESTS)
    message(FATAL_ERROR "NRlarge verification requires an existing Python environment with NumPy/SciPy")
  else()
    message(STATUS "NRlarge Python suites NOT enabled: Python/NumPy/SciPy unavailable; no dependencies installed")
  endif()
endif()
