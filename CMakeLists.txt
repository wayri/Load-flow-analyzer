cmake_minimum_required(VERSION 3.16)
project(LoadFlowAnalyzer VERSION 3.0 LANGUAGES CXX)
add_executable(load_flow Gauss_Siedel_Load_Flow_Analyser/Gauss_Siedel_Load_Flow_Analyser.cpp)
target_compile_features(load_flow PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(load_flow PRIVATE /W4 /permissive-)
else()
  target_compile_options(load_flow PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_solver tests/test_solver.cpp)
  target_compile_features(test_solver PRIVATE cxx_std_17)
  add_test(NAME numerical_solver COMMAND test_solver)
  add_executable(test_io tests/test_io.cpp)
  target_compile_features(test_io PRIVATE cxx_std_17)
  add_test(NAME csv_and_reports COMMAND test_io)
endif()
